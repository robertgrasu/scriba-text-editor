cmake_minimum_required(VERSION 3.20)
project(ScribaCpp C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# tree-sitter (MIT) and the grammars of the highlighted languages: plain C, compiled once into a static library
include(third_party/ts/sources.cmake)
add_library(tslangs STATIC third_party/ts/core/src/lib.c ${TS_LANG_SOURCES})
target_include_directories(tslangs PUBLIC third_party/ts/core/include)
target_include_directories(tslangs PRIVATE third_party/ts/core/src)
if(MSVC)
  target_compile_options(tslangs PRIVATE /O1 /w /utf-8 /Zi /std:c11)
endif()

add_executable(Scriba WIN32 src/main.cpp res/scriba.rc)
# the resource compiler does not report the files it embeds: without this a new icon or picture does not reach the exe
file(GLOB SCRIBA_RES_FILES res/*.ico res/hud/*.png)
set_source_files_properties(res/scriba.rc PROPERTIES OBJECT_DEPENDS "${SCRIBA_RES_FILES}")
target_compile_definitions(Scriba PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
# the scripted tests run without a license; a build that is given to anybody must be made with -DSCRIBA_TEST_BYPASS=OFF
option(SCRIBA_TEST_BYPASS "let the scripted tests skip the license" ON)
if(SCRIBA_TEST_BYPASS)
  target_compile_definitions(Scriba PRIVATE SCRIBA_TEST_BYPASS)
endif()
if(MSVC)
  target_compile_options(Scriba PRIVATE /utf-8 /W3 /O2 /bigobj)
endif()
# the C runtime goes into the exe: it starts on a computer that has no Visual C++ redistributable (both targets must agree)
set_property(TARGET Scriba tslangs PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  target_link_options(Scriba PRIVATE /MAP /DEBUG /PDBALTPATH:Scriba.pdb)
  target_compile_options(Scriba PRIVATE /Zi)
endif()
target_link_libraries(Scriba PRIVATE tslangs d2d1 dwrite user32 gdi32 comdlg32 shell32 dwmapi ole32 shlwapi windowscodecs)
